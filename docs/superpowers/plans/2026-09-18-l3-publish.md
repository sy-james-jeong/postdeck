# L3 Publish Orchestration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: superpowers:subagent-driven-development. Steps use `- [ ]` checkboxes.

**Goal:** Add a `publish(id)` seam to every adapter (flip draft→published in the source, round-trip safe) and a `postdeck publish` CLI that optionally deploys via `git push` (behind `--deploy`).

**Architecture:** `BlogSource.publish(id)` flips the source's draft marker (markdown/astro: patch frontmatter via FileStore, which auto-commits; notion: PATCH the Status property). A pure `buildDeployCommands` + a thin `deployProject` runner do the git push; the CLI ties publish + optional deploy together. The go-live git push is behind an explicit `--deploy` flag — the default is a local, reversible flip.

**Tech Stack:** TypeScript (NodeNext, strict, verbatimModuleSyntax), Vitest, node:child_process (git).

## Global Constraints
- `@postdeck/core` stays pure (zod-only). `.js` import extensions. Tests colocated. Additive: all 86 existing tests stay green.
- **No real `git push` in tests.** The git-executing `deployProject` is verified only by its pure `buildDeployCommands` helper + manual/real-env use.
- **`--deploy` gates the only outward-facing action** (git push → live). Default = flip only (local commit at most).
- Publish is round-trip safe: patch only the status/draft field, preserve all other frontmatter + body.

- [ ] **Step 3: Implement markdown `publish()`** — replace the stub in `packages/adapters/src/markdown.ts`:

```ts
    async publish(id: string): Promise<Ref> {
      const files = (await fs.list(dir)).filter((f) => f.endsWith('.md'))
      for (const f of files) {
        const content = await fs.read(`${dir}/${f}`)
        const raw = parseFrontmatter(content).data
        const slug = String(raw[fm.slug ?? 'slug'] ?? basename(f, '.md'))
        if (slug !== id) continue
        const patched = patchFrontmatter(content, { [fm.status ?? 'status']: cfg.publishStatus ?? 'published' })
        const path = `${dir}/${f}`
        await fs.write(path, patched, { message: `blog: publish ${slug}` })
        return { id: slug, path }
      }
      throw new Error(`markdown publish: no post with id "${id}" in ${dir}`)
    },
```

    async publish(id: string): Promise<Ref> {
      const slash = id.indexOf('/')
      const lang = slash >= 0 ? id.slice(0, slash) : src.langs[0]
      const slug = slash >= 0 ? id.slice(slash + 1) : id
      for (const ext of ['.md', '.mdx']) {
        const path = `${src.dir}/${lang}/${slug}${ext}`
        let content: string
        try {
          content = await fs.read(path)
        } catch (e: any) {
          if (e?.code === 'ENOENT') continue
          throw e
        }
        const patched = patchFrontmatter(content, { [fm.draft ?? 'draft']: false })
        await fs.write(path, patched, { message: `blog: publish ${lang}/${slug}` })
        return { id: `${lang}/${slug}`, path }
      }
      throw new Error(`astro publish: no post for id "${id}" in ${src.dir}`)
    },

- [ ] **Step 3: Implement `packages/adapters/src/deploy.ts`**

```ts
import { execFile, execFileSync } from 'node:child_process'
import { promisify } from 'node:util'

const run = promisify(execFile)

// Pure: the git command argv arrays a deploy runs, in order.
export function buildDeployCommands(opts: { emptyCommit?: boolean; message?: string }): string[][] {
  const cmds: string[][] = []
  if (opts.emptyCommit) cmds.push(['commit', '--allow-empty', '-m', opts.message ?? 'publish'])
  cmds.push(['push'])
  return cmds
}

export function gitToplevel(dir: string): string {
  return execFileSync('git', ['rev-parse', '--show-toplevel'], { cwd: dir, encoding: 'utf8' }).trim()
}

export async function deployProject(repoDir: string, opts: { emptyCommit?: boolean; message?: string }): Promise<void> {
  for (const args of buildDeployCommands(opts)) {
    await run('git', args, { cwd: repoDir })
  }
}
```

- [ ] **Step 5: Implement `apps/cli/src/publish.ts`**

```ts
import { resolve } from 'node:path'
import { config as loadDotenv } from 'dotenv'
import { resolveSource } from '@postdeck/core'
import { resolveConfigPath, loadBlogsConfig, createLocalFs, deployProject, gitToplevel } from '@postdeck/adapters'

export interface PublishArgs {
  project?: string
  slug?: string
  deploy: boolean
  config?: string
}

export function parsePublishArgs(argv: string[]): PublishArgs {
  const out: PublishArgs = { project: undefined, slug: undefined, deploy: false, config: undefined }
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i]
    if (a === '--project') out.project = argv[++i]
    else if (a === '--slug') out.slug = argv[++i]
    else if (a === '--deploy') out.deploy = true
    else if (a === '--config') out.config = argv[++i]
  }
  return out
}

export async function runPublish(argv: string[]): Promise<void> {
  const args = parsePublishArgs(argv)
  const cwd = process.cwd()
  loadDotenv({ path: resolve(cwd, '.env') })

  if (!args.project || !args.slug) {
    console.error('postdeck publish: --project <id> and --slug <slug> are required')
    process.exit(1)
  }

  const config = await loadBlogsConfig(resolveConfigPath(cwd, args.config))
  const cfg = config.find((b) => b.id === args.project)
  if (!cfg) {
    console.error(`postdeck publish: no project "${args.project}" in config. Available: ${config.map((b) => b.id).join(', ')}`)
    process.exit(1)
  }

  const env = (n: string) => process.env[n]
  const source = resolveSource(cfg, { fileStore: createLocalFs('/'), env, fetchImpl: fetch })
  const raws = await source.list()
  const target = raws.find((r) => r.slug === args.slug || r.id === args.slug)
  if (!target) {
    console.error(`postdeck publish: no post with slug "${args.slug}" in project "${args.project}"`)
    process.exit(1)
  }

  const ref = await source.publish(target.id)
  console.log(`published: ${ref.path ?? ref.url ?? ref.id}`)

  if (!args.deploy) {
    console.log('(local flip only — pass --deploy to git push and go live)')
    return
  }

  // Deploy = git push. For file-based sources the flip already committed; for
  // notion the flip touched no git, so make an empty commit to trigger CI.
  const isFile = cfg.source.type === 'markdown' || cfg.source.type === 'astro-collection'
  const repoDir = cfg.repoDir ?? (isFile ? gitToplevel((cfg.source as { dir: string }).dir) : undefined)
  if (!repoDir) {
    console.error(`postdeck publish: --deploy needs a git repo. Set "repoDir" for project "${args.project}".`)
    process.exit(1)
  }
  await deployProject(repoDir, { emptyCommit: !isFile, message: `publish: ${args.slug}` })
  console.log(`deployed: git push (${repoDir})`)
}
```
